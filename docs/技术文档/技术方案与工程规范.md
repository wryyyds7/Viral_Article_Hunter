## 一、技术栈选型

### 1.1 为什么选 Python 后端

| 考量 | Python | Java |
|------|--------|------|
| 团队门槛 | 学生团队上手快 | 学习曲线陡 |
| LLM集成 | openai/langchain生态成熟 | 需要额外封装 |
| 数据处理 | pandas/jieba天然优势 | 需要更多库 |
| 快速迭代 | Flask/FastAPI轻量灵活 | Spring Boot偏重 |
| 部署简单 | 单进程即可运行 | 需要JVM+Servlet容器 |

**结论：选Python，框架用FastAPI。** 理由：异步支持好（采集/改写都是IO密集）、自带API文档（Swagger）、类型提示强、性能足够。

### 4.3 service_base.py — Service基类

```python
"""
Service层基类，封装通用CRUD操作。
子类只需指定model_class，即可获得完整的CRUD能力。
"""
from typing import TypeVar, Generic, Optional, Type
from uuid import UUID
from sqlalchemy import select, func, update
from sqlalchemy.ext.asyncio import AsyncSession
from app.base.model_base import Base, ModelBase
from app.base.schema_base import PaginationParams, PaginationMeta

ModelType = TypeVar("ModelType", bound=Base)


class ServiceBase(Generic[ModelType]):
    """
    通用CRUD Service基类。
    
    使用方式：
        class ArticleService(ServiceBase[Article]):
            model_class = Article
    """

    model_class: Type[ModelType]

    def __init__(self, session: AsyncSession):
        self.session = session

    async def get_by_id(self, id: str) -> Optional[ModelType]:
        """按ID查询（排除已软删除的）"""
        stmt = select(self.model_class).where(
            self.model_class.id == id,
            self.model_class.is_deleted == False
        )
        result = await self.session.execute(stmt)
        return result.scalar_one_or_none()

    async def get_list(
        self,
        pagination: PaginationParams,
        filters: list = None,
        sort_by: str = "created_at",
        sort_order: str = "desc"
    ) -> tuple[list[ModelType], PaginationMeta]:
        """
        分页查询列表。
        返回 (数据列表, 分页元信息)
        """
        # 查询条件
        conditions = [self.model_class.is_deleted == False]
        if filters:
            conditions.extend(filters)

        # 总数
        count_stmt = select(func.count()).select_from(self.model_class).where(*conditions)
        total = (await self.session.execute(count_stmt)).scalar()

        # 排序
        order_col = getattr(self.model_class, sort_by, self.model_class.created_at)
        order = order_col.desc() if sort_order == "desc" else order_col.asc()

        # 分页
        stmt = (
            select(self.model_class)
            .where(*conditions)
            .order_by(order)
            .offset((pagination.page - 1) * pagination.page_size)
            .limit(pagination.page_size)
        )
        result = await self.session.execute(stmt)
        items = result.scalars().all()

        meta = PaginationMeta(
            page=pagination.page,
            page_size=pagination.page_size,
            total=total,
            total_pages=(total + pagination.page_size - 1) // pagination.page_size
        )
        return list(items), meta

    async def create(self, obj: ModelType) -> ModelType:
        """创建"""
        self.session.add(obj)
        await self.session.flush()
        return obj

    async def update(self, id: str, **kwargs) -> Optional[ModelType]:
        """按ID更新指定字段"""
        obj = await self.get_by_id(id)
        if not obj:
            return None
        for key, value in kwargs.items():
            setattr(obj, key, value)
        await self.session.flush()
        return obj

    async def soft_delete(self, id: str) -> bool:
        """软删除"""
        obj = await self.get_by_id(id)
        if not obj:
            return False
        obj.soft_delete()
        await self.session.flush()
        return True

    async def bulk_soft_delete(self, ids: list[str]) -> int:
        """批量软删除，返回删除数量"""
        stmt = (
            update(self.model_class)
            .where(self.model_class.id.in_(ids))
            .values(is_deleted=True)
        )
        result = await self.session.execute(stmt)
        return result.rowcount
```

```python
"""
API Key 加密存储工具。
使用 AES-GCM 对称加密，密钥从环境变量读取。
"""
import os
import base64
from cryptography.hazmat.primitives.ciphers.aead import AESGCM


class CryptoUtil:
    _key: bytes = b""

    @classmethod
    def init(cls, secret_key: str = None):
        """初始化加密密钥（应用启动时调用一次）"""
        key_str = secret_key or os.getenv("ENCRYPT_SECRET_KEY", "hotcontent-hunter-default-key-32byte")
        cls._key = key_str.encode("utf-8")[:32].ljust(32, b'\0')

    @classmethod
    def encrypt(cls, plaintext: str) -> str:
        """加密：返回 base64 编码的 nonce+ciphertext"""
        if not cls._key:
            cls.init()
        nonce = os.urandom(12)
        aesgcm = AESGCM(cls._key)
        ciphertext = aesgcm.encrypt(nonce, plaintext.encode("utf-8"), None)
        return base64.b64encode(nonce + ciphertext).decode("utf-8")

    @classmethod
    def decrypt(cls, encrypted: str) -> str:
        """解密：输入 base64 编码的 nonce+ciphertext"""
        if not cls._key:
            cls.init()
        raw = base64.b64decode(encrypted)
        nonce, ciphertext = raw[:12], raw[12:]
        aesgcm = AESGCM(cls._key)
        plaintext = aesgcm.decrypt(nonce, ciphertext, None)
        return plaintext.decode("utf-8")

    @classmethod
    def mask(cls, key: str) -> str:
        """掩码显示：sk-****xxxx（仅显示后4位）"""
        if len(key) <= 4:
            return "****"
        return f"****{key[-4:]}"
```

```python
DEFAULT_TIMEOUT = 15.0

DEFAULT_RETRY_WAIT = 3  # 秒


class HttpClient:

        self.timeout = timeout
        self.max_retries = max_retries

        return resp.json()

        stop=stop_after_attempt(DEFAULT_RETRY),
```

### 5.4 text_processor.py — 文本处理工具

```python
"""
文本处理工具集。
- 中文分词（jieba）
- 敏感词过滤
- 文本相似度计算
- 摘要截断
"""
import re
import jieba
import difflib
from pathlib import Path


class TextProcessor:
    """文本处理工具类"""

    _sensitive_words: set = set()

    @classmethod
    def load_sensitive_words(cls, filepath: str = "data/sensitive_words.txt"):
        """加载敏感词库"""
        path = Path(filepath)
        if path.exists():
            cls._sensitive_words = set(path.read_text(encoding="utf-8").strip().splitlines())
        else:
            cls._sensitive_words = set()

    @classmethod
    def filter_sensitive(cls, text: str, replace: str = "**") -> str:
        """过滤敏感词，替换为指定字符"""
        for word in cls._sensitive_words:
            if word in text:
                text = text.replace(word, replace * len(word))
        return text

    @staticmethod
    def segment(text: str) -> list[str]:
        """中文分词，返回词语列表"""
        return list(jieba.cut(text))

    @staticmethod
    def extract_keywords(text: str, top_k: int = 5) -> list[str]:
        """提取关键词"""
        import jieba.analyse
        return jieba.analyse.extract_tags(text, topK=top_k)

    @staticmethod
    def similarity(text1: str, text2: str) -> float:
        """
        计算两段文本的相似度（0-1）。
        使用SequenceMatcher，简单快速。
        """
        return difflib.SequenceMatcher(None, text1, text2).ratio()

    @staticmethod
    def truncate(text: str, max_len: int = 200, suffix: str = "...") -> str:
        """截断文本，保留最大长度"""
        if len(text) <= max_len:
            return text
        return text[:max_len - len(suffix)] + suffix

    @staticmethod
    def count_words(text: str) -> int:
        """统计中文字数（去除空白和标点）"""
        chinese_chars = re.findall(r'[\u4e00-\u9fff]', text)
        return len(chinese_chars)
```

```python
    - 互动钩子 25%

        "title_power": 0.30,
        "emotion_density": 0.25,
        "interaction_hook": 0.25,

        title_power: int,
        emotion_density: int,

        计算综合评分。

            "interaction_hook": min(100, max(0, interaction_hook)),
            "structure_density": min(100, max(0, structure_density)),
```

```python
    @classmethod
    def recognize(cls, title: str, max_results: int = 2) -> list[dict]:
        """
        识别标题中使用的公式。
        返回按置信度排序的结果列表。
        """
        results = []
        for formula_type, rule in cls.RULES.items():
            confidence = 0.0
            matched_text = ""

            # 正则匹配
            for pattern in rule["patterns"]:
                match = re.search(pattern, title)
                if match:
                    confidence = max(confidence, 0.8)
                    matched_text = match.group()
                    break

            # 关键词匹配
            for kw in rule["keywords"]:
                if kw in title:
                    confidence = max(confidence, 0.6)
                    if not matched_text:
                        matched_text = kw
                    break

            if confidence > 0:
                results.append({
                    "type": formula_type.value,
                    "confidence": round(confidence, 2),
                    "matched_text": matched_text
                })

        # 按置信度降序排列，取前N个
        results.sort(key=lambda x: x["confidence"], reverse=True)
        return results[:max_results]
```

```python
    文件名与Platform枚举值对应（如xiaohongshu.md）。

    def load(cls, platform: Platform) -> str:
        """加载指定平台的改写规则文本"""

        if key in cls._cache:
            return cls._cache[key]

            content = filepath.read_text(encoding="utf-8")
        else:
            content = f"平台 {key} 的改写规则文件不存在，请创建 data/platform_rules/{key}.md"

        return result

    @classmethod
```

### 5.9 exporter.py — 导出工具

```python
"""
导出工具。
支持 MD / Word / ZIP 格式导出。
"""
import io
import json
import zipfile
from datetime import datetime
from docx import Document
from docx.shared import Pt


class Exporter:
    """内容导出器"""

    @staticmethod
    def to_markdown(title: str, content: str, tags: list[str], meta: dict = None) -> str:
        """生成Markdown格式内容，带YAML Front Matter"""
        front_matter = {
            "title": title,
            "tags": tags,
            "exported_at": datetime.now().isoformat(),
        }
        if meta:
            front_matter.update(meta)

        lines = ["---"]
        lines.append(json.dumps(front_matter, ensure_ascii=False, indent=2))
        lines.append("---")
        lines.append("")
        lines.append(f"# {title}")
        lines.append("")
        lines.append(content)
        lines.append("")
        if tags:
            lines.append(" ".join(f"#{t}" for t in tags))

        return "\n".join(lines)

    @staticmethod
    def to_docx(title: str, content: str, tags: list[str]) -> bytes:
        """生成Word文档，返回字节流"""
        doc = Document()
        doc.add_heading(title, level=1)

        # 正文按段落写入
        for para in content.split("\n"):
            if para.strip():
                p = doc.add_paragraph(para.strip())
                for run in p.runs:
                    run.font.size = Pt(12)

        # 标签
        if tags:
            doc.add_paragraph("")
            doc.add_paragraph("标签：" + "、".join(tags))

        buffer = io.BytesIO()
        doc.save(buffer)
        return buffer.getvalue()

    @staticmethod
    def to_zip(files: dict[str, bytes], filename_prefix: str = "爆文猎人") -> bytes:
        """
        打包为ZIP文件。
        files: {文件名: 字节内容}
        """
        buffer = io.BytesIO()
        with zipfile.ZipWriter(buffer, 'w', zipfile.ZIP_DEFLATED) as zf:
            for name, data in files.items():
                zf.writestr(name, data)

        return buffer.getvalue()

    @staticmethod
    def batch_export_markdown(results: dict[str, dict]) -> bytes:
        """
        批量导出多平台改写结果为ZIP。
        results: {platform: {title, content, tags}}
        """
        files = {}
        for platform, data in results.items():
            md = Exporter.to_markdown(
                title=data["title"],
                content=data["content"],
                tags=data.get("tags", []),
                meta={"platform": platform}
            )
            files[f"{platform}.md"] = md.encode("utf-8")

        return Exporter.to_zip(files, filename_prefix="爆文猎人_改写结果")
```

### 5.10 rate_limiter.py — 限流器

```python
"""
内存限流器。
基于滑动窗口计数，满足接口文档中的限流要求。
"""
import time
from collections import defaultdict


class RateLimiter:
    """
    滑动窗口限流器。
    用法：
        limiter = RateLimiter()
        if not limiter.allow("user:123:collect", max_requests=10, window_seconds=60):
            raise AppException(42901, "操作过于频繁")
    """

    _requests: dict[str, list[float]] = defaultdict(list)

    @classmethod
    def allow(cls, key: str, max_requests: int, window_seconds: int = 60) -> bool:
        """
        判断是否允许请求。
        key: 限流维度（如 "user:123:collect"）
        max_requests: 窗口内最大请求数
        window_seconds: 窗口时间（秒）
        """
        now = time.time()
        window_start = now - window_seconds

        # 清理过期记录
        cls._requests[key] = [
            t for t in cls._requests[key] if t > window_start
        ]

        if len(cls._requests[key]) >= max_requests:
            return False

        cls._requests[key].append(now)
        return True

    @classmethod
    def remaining(cls, key: str, max_requests: int, window_seconds: int = 60) -> int:
        """返回剩余可用请求次数"""
        now = time.time()
        window_start = now - window_seconds
        current = len([t for t in cls._requests[key] if t > window_start])
        return max(0, max_requests - current)

    @classmethod
    def clear(cls, key: str = None):
        """清除限流记录"""
        if key:
            cls._requests.pop(key, None)
        else:
            cls._requests.clear()
```

### 12.5 事件总线代码骨架

```python
# app/services/event_bus.py
import asyncio
from collections import defaultdict
from typing import Any, Callable, Coroutine
from dataclasses import dataclass
import logging

logger = logging.getLogger(__name__)


@dataclass
class Event:
    """事件对象"""
    name: str
    payload: dict[str, Any]
    source: str  # 生产者标识


class EventBus:
    """基于异步队列的发布-订阅事件总线"""

    def __init__(self):
        self._subscribers: dict[str, list[Callable[[Event], Coroutine]]] = defaultdict(list)

    def subscribe(self, event_name: str, handler: Callable[[Event], Coroutine]):
        """订阅事件"""
        self._subscribers[event_name].append(handler)
        logger.info(f"订阅事件: {event_name} -> {handler.__name__}")

    async def publish(self, event: Event):
        """发布事件，所有订阅者异步执行"""
        handlers = self._subscribers.get(event.name, [])
        if not handlers:
            logger.debug(f"事件 {event.name} 无订阅者")
            return

        tasks = [self._safe_call(handler, event) for handler in handlers]
        await asyncio.gather(*tasks)

    async def _safe_call(self, handler: Callable[[Event], Coroutine], event: Event):
        """安全调用处理器，异常不影响其他订阅者"""
        try:
            await handler(event)
        except Exception as e:
            logger.error(
                f"事件处理器异常: event={event.name}, "
                f"handler={handler.__name__}, error={e}"
            )


# 全局单例
event_bus = EventBus()


# --- 订阅注册示例 ---

async def on_collect_completed(event: Event):
    """采集完成 → 自动触发分析"""
    from app.services.analysis_service import AnalysisService
    service = AnalysisService()
    await service.batch_analyze(event.payload["articles"])


async def on_upload_completed(event: Event):
    """上传完成 → 检查阈值 → 触发分析"""
    from app.services.analysis_service import AnalysisService
    payload = event.payload

    if payload["success_count"] < UPLOAD_THRESHOLD:
        # 发布阈值不足事件，由前端弹窗处理
        await event_bus.publish(Event(
            name="upload.threshold_not_met",
            payload=payload,
            source="upload_service"
        ))
    else:
        service = AnalysisService()
        await service.batch_analyze_by_batch(payload["batch_id"])


# 启动时注册
event_bus.subscribe("collect.completed", on_collect_completed)
event_bus.subscribe("upload.completed", on_upload_completed)
```
